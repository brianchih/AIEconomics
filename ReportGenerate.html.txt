<script>
    // 🔑 請在此處確認填入您的 32 位元 FRED API Key
    const API_KEY = 'YOUR_FRED_API_KEY'; 

    const SERIES_ID = 'CPIAUCSL';
    const debugLog = document.getElementById('debugLog');

    function log(msg) {
        debugLog.innerHTML += msg + '<br>';
    }

    async function fetchUSData() {
        const cleanKey = API_KEY.trim();
        log(`🔑 API Key 字元數檢查：<strong>${cleanKey.length}</strong> 字元`);
        
        if (!cleanKey || cleanKey === 'YOUR_FRED_API_KEY' || cleanKey.length !== 32) {
            log('❌ <span class="badge-danger">設定錯誤</span>：請確認已在代碼中填入長度正確的 32 位元 API Key。');
            return null;
        }

        const rawUrl = `https://api.stlouisfed.org/fred/series/observations?series_id=${SERIES_ID}&api_key=${cleanKey}&file_type=json&units=pc1&sort_order=desc&limit=10`;
        
        // 多重雙向通道（備用 Proxy 輪詢機制）
        const proxies = [
            `https://api.allorigins.win/get?url=${encodeURIComponent(rawUrl)}`,
            `https://corsproxy.io/?${encodeURIComponent(rawUrl)}`
        ];

        for (let i = 0; i < proxies.length; i++) {
            const proxyUrl = proxies[i];
            log(`🌐 嘗試透過 HTTP 通道 ${i + 1} 抓取 FRED 資料...`);

            try {
                const res = await fetch(proxyUrl);
                if (!res.ok) continue;

                let jsonRes = await res.json();
                
                // 相容 allorigins 的 contents 包裹結構
                if (jsonRes.contents) {
                    jsonRes = JSON.parse(jsonRes.contents);
                }

                if (jsonRes.error_code) {
                    log(`❌ <span class="badge-danger">FRED API 錯誤 (${jsonRes.error_code})</span>：${jsonRes.error_message}`);
                    return null;
                }

                if (jsonRes && jsonRes.observations && jsonRes.observations.length > 0) {
                    log('✅ <span class="badge-success">連線成功！</span>已從 FRED 官方成功取得數據。');
                    return jsonRes.observations
                        .filter(o => o.value !== "." && !isNaN(parseFloat(o.value)))
                        .map(o => ({
                            date: o.date,
                            value: parseFloat(o.value).toFixed(2)
                        }));
                }
            } catch (err) {
                log(`⚠️ 通道 ${i + 1} 嘗試失敗，自動切換至下一個通道...`);
            }
        }

        log('❌ <span class="badge-danger">連線失敗</span>：無法透過代理通道連線至 FRED API。');
        return null;
    }

    async function init() {
        const latestValEl = document.getElementById('latestVal');
        const latestDateEl = document.getElementById('latestDate');
        const tableBody = document.getElementById('seriesTable');

        const observations = await fetchUSData();

        if (observations && observations.length > 0) {
            const latest = observations[0];
            latestValEl.classList.remove('loading');
            latestValEl.innerText = `${latest.value}%`;
            latestDateEl.innerText = `最新統計基期：${latest.date} (FRED 官方 API)`;

            tableBody.innerHTML = observations.map((item, index) => `
                <tr>
                    <td><strong>${item.date}</strong> ${index === 0 ? '<span style="color:#2563eb; font-size:0.8rem; margin-left:4px;">(最新期)</span>' : ''}</td>
                    <td><span style="font-weight:bold; color:#2563eb;">${item.value}%</span></td>
                    <td>FRED 官方 API</td>
                </tr>
            `).join('');
        } else {
            latestValEl.classList.remove('loading');
            latestValEl.style.color = '#ef4444';
            latestValEl.innerText = '讀取失敗';
            latestDateEl.innerText = '請查看上方黑色區塊的診斷訊息';
            tableBody.innerHTML = `<tr><td colspan="3" style="text-align:center; color:#ef4444;">連線失敗，請參考診斷日誌排查</td></tr>`;
        }
    }

    window.onload = init;
</script>
